import { useState, type FormEvent } from "react";
import { available } from "../data/puppies";
import { ADDRESS, EMAIL, PHONE_DISPLAY, PHONE_TEL, WA_LINK } from "../data/site";
import { usePageMeta } from "../lib/usePageMeta";

export default function Contact() {
  usePageMeta("Visit & contact | TeddyPups U.A.E.", "Visit our Downtown Dubai showroom or send us a letter.");
  const [sent, setSent] = useState(false);
  const [err, setErr] = useState("");

  function submit(e: FormEvent<HTMLFormElement>) {
    e.preventDefault();
    const f = new FormData(e.currentTarget);
    if (f.get("website")) return;
    if (String(f.get("name") || "").trim().length < 2 || String(f.get("phone") || "").trim().length < 7) {
      setErr("Please add your name and a phone number so we can reply.");
      return;
    }
    setErr("");
    setSent(true);
  }

  return (
    <div className="page page--warm">
      <section className="page-hero gutter">
        <h1 className="page-title">Get in touch with us</h1>
        <p className="page-lede">Have questions about our puppies or adoption process? Fill out the form, and our team will get back to you promptly. Whether ready to adopt or just seeking info, we'd love to hear from you!</p>
      </section>
      <section className="gutter tp-contactgrid">
        {sent ? (
          <div className="tp-letter tp-letter--sent" role="status">
            <p className="tp-form__thanks">Letter posted! 💌</p>
            <p>Thank you — we'll be in touch very soon.</p>
          </div>
        ) : (
          <form className="tp-letter" onSubmit={submit} noValidate>
            <p>Dear TeddyPups,</p>
            <p>
              my name is <input name="name" aria-label="Your name" placeholder="your name" /> , and you can reach me on{" "}
              <input name="phone" type="tel" aria-label="Phone" placeholder="phone / WhatsApp" /> or by email at{" "}
              <input name="email" type="email" aria-label="Email" placeholder="you@email.com" />. I live in{" "}
              <input name="area" aria-label="Area" placeholder="your area" />, and I'm curious about{" "}
              <select name="puppy" aria-label="Puppy you're interested in" defaultValue="">
                <option value="">any puppy (not sure yet)</option>
                {available.map((p) => (
                  <option key={p.slug} value={p.slug}>{p.name} ({p.breed === p.name ? "" : p.breed + ", "}{p.sex})</option>
                ))}
              </select>.
            </p>
            <label className="tp-field">Your message (optional)<textarea name="message" rows={3} /></label>
            <label className="tp-hp" aria-hidden="true">Website<input name="website" tabIndex={-1} autoComplete="off" /></label>
            {err && <p className="tp-field__err" role="alert">{err}</p>}
            <p className="tp-letter__sign">with love, a future puppy parent</p>
            <button className="btn btn-primary" type="submit">Post my letter</button>
          </form>
        )}
        <div className="tp-contactcards">
          <a className="tp-card" href={WA_LINK} target="_blank" rel="noreferrer"><small>WhatsApp</small>{PHONE_DISPLAY}</a>
          <a className="tp-card" href={PHONE_TEL}><small>call us</small>{PHONE_DISPLAY}</a>
          <a className="tp-card" href={`mailto:${EMAIL}`}><small>email</small>{EMAIL}</a>
        </div>
      </section>
      <section className="gutter tp-visit">
        <h2 className="section-title">Visit us in Downtown Dubai</h2>
        <p className="page-lede">Step into our Downtown Dubai showroom and experience the joy of choosing your puppy face-to-face.</p>
        <p>{ADDRESS} · <a className="link-ink" href="https://www.instabizweb.com/" target="_blank" rel="noreferrer">Open in Google Maps</a></p>
        <img src="/media/skvodl2NZiKq/cefab5eee9/27b888_323008bb0f6a41d4bde2a0da7206cd98f003.jpg" alt="The apricot teddy bear puppy from our showroom, held up to say hello" loading="lazy" />
      </section>
    </div>
  );
}
